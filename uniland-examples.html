// Консольная (без окна) версия приветствия: спрашивает имя и отвечает.
// Запуск:  uniland run examples/greet_cli.uni

wordcount.uniПодсчёт слов — Файлы, строки, массивы и функции в одном маленьком скрипте.
// Считает слова, строки и самые частые слова в текстовом файле.
// Запуск:  uniland run examples/wordcount.uni README.md

let files = argv()
if (length(files) == 0) {
    print("Использование: uniland run examples/wordcount.uni <файл>")
    exit(1)
}

let path = files[0]
if (!file_exists(path)) {
    print(`Файл не найден: {path}`)
    exit(1)
}

let text = read_file(path)
let lines = split(text, "\n")
let words = split(lowercase(text), " ")

// частота слов через объект
let freq = {}
for (w in words) {
    let key = trim(w)
    if (length(key) > 0) {
        if (has(freq, key)) { freq[key] += 1 } else { freq[key] = 1 }
    }
}

print(`Строк:  {length(lines)}`)
print(`Слов:   {length(words)}`)
print(`Разных: {length(keys(freq))}`)
ticker.uniТикер — Долгоживущий скрипт: цикл, пауза, обновление вывода.
// Долгоживущий скрипт: тикает раз в секунду и не завершается,
// пока ты не нажмёшь Ctrl+C. Так делают простые службы и боты.
// Запуск:  uniland run examples/ticker.uni

let tick = 0
log_info("Сервис запущен. Останови меня через Ctrl+C.")

while (true) {
    tick += 1
    log_info(`тик #{tick} — время: {format_time(get_time(), "%H:%M:%S")}`)
    sleep(1000)   // пауза в миллисекундах
}
gui_demo.uniОконная программа — GUI на tkinter прямо из UniLand.
// Маленькое окно на UniLand: вводишь имя — получаешь приветствие.
// Запуск:  uniland run examples/gui_demo.uni

let win = gui_window("Привет, UniLand", 320, 160)
gui_label(win, "Как тебя зовут?")
let field = gui_input(win, "")
let out = gui_label(win, "")

gui_button(win, "Поздороваться", () => {
    let name = gui_get(field)
    if (length(name) == 0) {
        gui_set(out, "Введи имя :)")
    } else {
        gui_set(out, `Привет, {name}!`)
    }
})

gui_button(win, "Выход", () => { gui_close(win) })

gui_run(win)

// Демонстрация медиа и файлов в UniLand 2.2
// Запуск:  uniland run examples/media_demo.uni
//
// Положи рядом картинку pic.png (png/gif работают без Pillow).
// GIF и видео — по желанию (anim.gif, clip.mp4).

let win = gui_window("UniLand — медиа", 420, 560)
gui_label(win, "Картинка, кнопка-картинка, выбор файла и архивы:")

// показать картинку (если её нет — будет понятная ошибка)
try {
    gui_image(win, "pic.png", 120, 120)
} catch (e) {
    gui_label(win, "нет pic.png — положи картинку рядом")
}

let out = gui_label(win, "")

// кнопка-картинка: клик считает нажатия
let clicks = 0
try {
    gui_image_button(win, "pic.png", () => {
        clicks += 1
        gui_set(out, `Кликов: {clicks}`)
    }, 96, 96)
} catch (e) { }

// выбрать любой файл через системный диалог
gui_button(win, "Выбрать файл…", () => {
    let f = pick_file("Выбери что-нибудь")
    if (f == null) {
        gui_set(out, "Отменено")
    } else {
        gui_set(out, `Выбран: {f}`)
    }
})

// открыть файл/ссылку в системном приложении
gui_button(win, "Открыть сайт языка", () => {
    open_url("https://codeberg.org/moyunni/uniland")
})

// заархивировать и показать содержимое
gui_button(win, "Сделать zip из этой папки", () => {
    zip(".", "demo.zip")
    gui_set(out, `В архиве: {join(zip_list("demo.zip"), ", ")}`)
})

// анимированный GIF (если есть)
try { gui_gif(win, "anim.gif", 160, 120) } catch (e) { }

gui_button(win, "Выход", () => { gui_close(win) })
gui_run(win)
selftest.uniСамопроверка языка — Набор проверок: синтаксис, встроенные функции, ошибки.
// ============================================================
//  UniLand self-test — проверяет интерпретатор сам на себе.
//  Запуск:  uniland run examples/selftest.uni
// ============================================================

let passed = 0
let failed = 0

func check(name, got, want) {
    if (got == want) {
        passed += 1
    } else {
        failed += 1
        print(`  FAIL: {name}  (получили {got}, ждали {want})`)
    }
}

// ---- арифметика и приоритеты ----
check("math", 2 + 3 * 4, 14)
check("power", 2 ^ 10, 1024)
check("mod", 17 % 5, 2)
check("unary", -(-5), 5)
check("compare", 3 < 5 && 5 <= 5, true)

// ---- строки и шаблоны ----
let who = "мир"
check("template", `привет, {who}!`, "привет, мир!")
check("concat", "a" + 1 + true, "a1true")
check("upper", uppercase("abc"), "ABC")
check("split", length(split("a,b,c", ",")), 3)
check("trim", trim("  hi  "), "hi")

// ---- функции, рекурсия, замыкания ----
func fact(n) {
    if (n <= 1) { return 1 }
    return n * fact(n - 1)
}
check("recursion", fact(6), 720)

func adder(step) { return x => x + step }
let add10 = adder(10)
check("closure", add10(5), 15)

// ---- условия и циклы ----
let total = 0
for (let i = 1; i <= 100; i++) { total += i }
check("for-classic", total, 5050)

let s = 0
for (n in [3, 4, 5]) { s += n }
check("for-in-array", s, 12)

let collected = ""
for (ch in "abc") { collected += ch }
check("for-in-string", collected, "abc")

let n = 0
while (n < 5) { n++ }
check("while", n, 5)

// ---- break / continue ----
let acc = 0
for (let i = 0; i < 10; i++) {
    if (i == 5) { break }
    if (i % 2 == 0) { continue }
    acc += i
}
check("break-continue", acc, 4)  // 1 + 3

// ---- массивы ----
let arr = [1, 2, 3]
push(arr, 4)
check("push", length(arr), 4)
check("pop", pop(arr), 4)
check("map", join(map([1, 2, 3], x => x * 2), ","), "2,4,6")
check("filter", length(filter([1, 2, 3, 4], x => x % 2 == 0)), 2)
check("reduce", reduce([1, 2, 3, 4], (a, b) => a + b, 0), 10)
check("sort", join(sort([3, 1, 2]), ""), "123")
check("includes", includes([1, 2, 3], 2), true)
check("range", length(range(0, 10)), 10)

// ---- объекты ----
let user = {name: "Аня", age: 20}
user.age += 1
user["city"] = "Питер"
check("obj-dot", user.name, "Аня")
check("obj-update", user.age, 21)
check("obj-index", user["city"], "Питер")
check("keys", length(keys(user)), 3)
check("has", has(user, "name"), true)

// ---- вложенные структуры ----
let grid = [[1, 2], [3, 4]]
grid[0][1] = 99
check("nested-index", grid[0][1], 99)

// ---- ошибки ----
let caught = "no"
try {
    throw "oops"
} catch (e) {
    caught = e
}
check("try-catch", caught, "oops")

// ---- встроенный Python ----
python {
    import math
    uni.sqrt2 = math.sqrt(2)
}
check("python-block", round(sqrt2, 5), 1.41421)
check("pyeval", pyeval("3 * 14"), 42)

// ---- JSON ----
let data = parse_json("{\"x\": 1, \"y\": [2, 3]}")
check("json-parse", data["y"][1], 3)

// ============================================================
print("")
print(`Готово: {passed} прошло, {failed} упало.`)
if (failed > 0) { exit(1) }
